Q1--------- What is the total ridership for each transport type year by year? 
-----------"ما هو اجمالي عدد الركاب لكل وسائل النقل كل عام؟"

select Year,
SUM(Subways_Total_Ridership) as 'Subways Passengers',
SUM(Buses_Total_Ridership) as 'Buses Passengers',
SUM(AAR_Total_Ridership) as 'AAR Passengers',
SUM(Bridges_Tunnels_Total_Ridership) as 'Bridges_Tunnels Passengers',
SUM(LIRR_Total_Ridership) as 'LIRR Passengers',
SUM(MNR_Total_Ridership) as 'MNR Passengers',
SUM(SIR_Total_Ridership) as 'SIR Passengers'
from MTA_Daily_Ridership
group by Year
order by Year asc

Q3 ----- Compare between the average weekday ridership vs average weekend ridership for each transportation mode? 
--------" قارن بين متوسط عدد الركاب خلال أيام الأسبوع و متوسط عدد الركاب خلال إجازة الأسبوع بالنسبة لكل وسائل النقل؟"

select 
AVG(Subways_Total_Ridership) as 'Average Subways Passengers',
AVG(Buses_Total_Ridership) as 'Average Buses Passengers',
AVG(AAR_Total_Ridership) as 'Average AAR Passengers',
AVG(Bridges_Tunnels_Total_Ridership) as 'Average Bridges_Tunnels Passengers',
AVG(LIRR_Total_Ridership) as 'Average LIRR Passengers',
AVG(MNR_Total_Ridership) as 'Average MNR Passengers',
AVG(SIR_Total_Ridership) as 'Average SIR Passengers',
case
when Day_Name in ('Saturday', 'Sunday') then 'Weekend'
when Day_Name in ('Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday') then 'Weekday'
else 'weekday'
end as 'Day_Type' 
from MTA_Daily_Ridership
group by case
when Day_Name in ('Saturday', 'Sunday') then 'Weekend'
when Day_Name in ('Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday') then 'Weekday'
else 'weekday'
end

Q8-----Which transportation mode recovered the fastest after the pandemic?
-------"ما هي وسيلة النقل التي تعافت بأسرع وتيرة بعد الجائحة؟"

select TOP 7 [Transport_Mode], AVG(Recovery_Percentage) AS Avg_Recovery
FROM (
    SELECT 'Subways' AS Transport_Mode, Subways_Prepandemic AS Recovery_Percentage FROM MTA_Daily_Ridership
    UNION ALL
    SELECT 'Buses' AS Transport_Mode, Buses_prepandemic FROM MTA_Daily_Ridership
    UNION ALL
	SELECT 'AAR' AS Transport_Mode, AAR_Prepandemic FROM MTA_Daily_Ridership
    UNION ALL
	SELECT 'Bridges_Tunnels' AS Transport_Mode, Bridges_Tunnels_Prepandemic FROM MTA_Daily_Ridership
    UNION ALL
    SELECT 'LIRR' AS Transport_Mode, LIRR_Prepandemic FROM MTA_Daily_Ridership
    UNION ALL
    SELECT 'MNR' AS Transport_Mode, MNR_Prepandemic FROM MTA_Daily_Ridership
	UNION ALL
	SELECT 'SIR' AS Transport_Mode, SIR_Prepandemic FROM MTA_Daily_Ridership
	)
 AS Recovery_Data
GROUP BY Transport_Mode
ORDER BY Avg_Recovery DESC

Q9-----Which transportation mode is most sensitive to the effects of the pandemic?
-------"ما هي وسيلة النقل الأكثر تأثراً بتداعيات الجائحة؟"

select TOP 1 [Transport_Mode], MIN (Avg_Recovery) As Lowest_Avg_Recovery
FROM (
    SELECT 'Subways' AS Transport_Mode, avg(Subways_Prepandemic) AS Avg_Recovery FROM MTA_Daily_Ridership
    UNION ALL
    SELECT 'Buses' AS Transport_Mode, avg(Buses_prepandemic) AS Avg_Recovery FROM MTA_Daily_Ridership
    UNION ALL
	SELECT 'AAR' AS Transport_Mode, avg(AAR_Prepandemic) AS Avg_Recovery FROM MTA_Daily_Ridership
    UNION ALL
	SELECT 'Bridges_Tunnels' AS Transport_Mode, avg(Bridges_Tunnels_Prepandemic) AS Avg_Recovery FROM MTA_Daily_Ridership
    UNION ALL
    SELECT 'LIRR' AS Transport_Mode, avg(LIRR_Prepandemic) AS Avg_Recovery FROM MTA_Daily_Ridership
    UNION ALL
    SELECT 'MNR' AS Transport_Mode, avg(MNR_Prepandemic) AS Avg_Recovery FROM MTA_Daily_Ridership
	UNION ALL
	SELECT 'SIR' AS Transport_Mode, avg(SIR_Prepandemic) AS Avg_Recovery FROM MTA_Daily_Ridership
	)
 AS Pandemic_Impact
GROUP BY Transport_Mode
ORDER BY Lowest_Avg_Recovery ASC
